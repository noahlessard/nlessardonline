export const buttonModules = import.meta.glob('../assets/buttons/*', { eager: true });
export const buttons = Object.values(buttonModules).map((m) => m.default);

export function chunk(arr, size) {
  const out = [];
  for (let i = 0; i < arr.length; i += size) out.push(arr.slice(i, i + size));
  return out;
}

export const buttonGroups = chunk(buttons, 5);
